-- 1: Database health check
SELECT 1

-- 2: Execute the Company Payroll Report stored procedure
EXEC dbo.usp_CompanyPayrollReport

-- 3: Approved TXT upload and employee payroll report
SELECT e.employee_code AS employee_id, e.employee_name, e.department, e.designation,
       p.payroll_month, p.basic_salary, p.allowances, p.deductions,
       p.net_salary, p.payment_status
FROM employees e
JOIN payroll p ON p.employee_id = e.id
WHERE e.company_id = :company_id
  AND p.payroll_month = (
      SELECT MAX(p2.payroll_month)
      FROM payroll p2
      JOIN employees e2 ON e2.id = p2.employee_id
      WHERE e2.company_id = :company_id
  )
ORDER BY e.employee_code

-- 5: Find login user by username and reload the committed profile
SELECT id, username, email, full_name, user_principal_name, given_name, surname,
       department, title, company, permissions, is_active, created_at, updated_at
FROM users
WHERE username = :username

)

    :username, :email, :full_name, :user_principal_name, :given_name, :surname,

-- 7: Update an authenticated user's profile and modification timestamp
UPDATE users
SET email = :email,
    full_name = :full_name,
    user_principal_name = :user_principal_name,
    given_name = :given_name,
    surname = :surname,
    department = :department,
    title = :title,
    company = :company,

-- 8: List or find companies, optionally restricted to selected IDs
SELECT id, company_name
FROM companies
WHERE (:record_id IS NULL OR id = :record_id)
  AND (:filter_company_ids = 0 OR id IN :company_ids)
ORDER BY company_name

-- 9: List clients for a company
SELECT id, company_id, client_code, client_name FROM clients WHERE company_id = :company_id ORDER BY client_name

-- 10: List email addresses for a client or company
SELECT ce.id, ce.client_id, ce.email
FROM client_emails ce
JOIN clients c ON c.id = ce.client_id
WHERE (:client_id IS NULL OR ce.client_id = :client_id)
  AND (:company_id IS NULL OR c.company_id = :company_id)
ORDER BY ce.email

-- 18: List execution history, find a run, or filter failures by cutoff
SELECT *
FROM job_runs
WHERE (:record_id IS NULL OR id = :record_id)
  AND (:failed_before IS NULL OR (status IN :statuses AND started_at <= :failed_before))
ORDER BY id DESC
